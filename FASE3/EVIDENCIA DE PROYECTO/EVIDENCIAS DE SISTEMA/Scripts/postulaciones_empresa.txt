    <script>
      const IS_EMPRESA = (document.body.dataset.role === 'empresa');

      document.addEventListener('click', (e) => {
        const card = e.target.closest('.postulante-item');
        if (card && card.dataset.id) {
          console.log('Click detectado en:', card);
          mostrarDetalle(card);
        }
      });

      function mostrarDetalle(el) {
        const isMobile = window.innerWidth <= 768;
        console.log('mostrarDetalle llamado, isMobile:', isMobile);
        
        document.querySelectorAll('.postulante-item.active').forEach(n => n.classList.remove('active'));
        el.classList.add('active');
        
        const d = el.dataset;
        const estadoClass = d.estado === 'pendiente' ? 'status-pendiente' : (d.estado === 'aceptado' ? 'status-aceptado' : 'status-rechazado');
        
        let acciones = '';
        if (d.cv) {
          acciones += '<a class="btn-s btn-cv" href="' + d.viewCvUrl + '" target="_blank" rel="noopener">Ver CV</a>';
          acciones += ' <a class="btn-s" style="background:#475569" href="' + d.viewCvUrl + '?download=1" target="_blank" rel="noopener">Descargar</a>';
        }
        else acciones += '<span style="color:#6b7280;">Sin CV</span>';
        
        if (IS_EMPRESA) {
          if (d.estado !== 'aceptado') acciones += ' <a class="btn-s btn-acc" href="' + d.acceptUrl + '">Aceptar</a>';
          if (d.estado !== 'rechazado') acciones += ' <a class="btn-s btn-rej" href="' + d.rejectUrl + '">Rechazar</a>';
        }
        
        const html =
          '<div class="detail-card">' +
            '<div class="info-grid">' +
              '<div class="info-row">' +
                '<div class="label">Postulante</div>' +
                '<div class="value">' + d.nombre + ' ' + d.apellido + '</div>' +
              '</div>' +
              '<div class="info-row">' +
                '<div class="label">RUT</div>' +
                '<div class="value">' + (d.rut || '-') + '</div>' +
              '</div>' +
              '<div class="info-row">' +
                '<div class="label">Teléfono</div>' +
                '<div class="value">' + (d.telefono || '-') + '</div>' +
              '</div>' +
              '<div class="info-row">' +
                '<div class="label">Correo</div>' +
                '<div class="value">' + d.correo + '</div>' +
              '</div>' +
              '<div class="info-row">' +
                '<div class="label">Institución</div>' +
                '<div class="value">' + (d.institucion || '-') + '</div>' +
              '</div>' +
              '<div class="info-row">' +
                '<div class="label">Anuncio</div>' +
                '<div class="value">' + d.anuncio + '</div>' +
              '</div>' +
              '<div class="info-row">' +
                '<div class="label">Fecha de Postulación</div>' +
                '<div class="value">' + d.fecha + '</div>' +
              '</div>' +
              '<div class="info-row">' +
                '<div class="label">Estado</div>' +
                '<div class="value ' + estadoClass + '">' + d.estado.charAt(0).toUpperCase() + d.estado.slice(1) + '</div>' +
              '</div>' +
            '</div>' +
            '<div class="actions">' + acciones + '</div>' +
          '</div>';
        
        if (isMobile) {
          // Obtener elementos del modal cada vez (porque están fuera del bloque if)
          const modalContent = document.getElementById('modalContentPostulacion');
          const modalOverlay = document.getElementById('modalOverlay');
          const modalBottom = document.getElementById('modalBottom');
          
          console.log('Elementos del modal:', {
            modalContent: modalContent,
            modalOverlay: modalOverlay,
            modalBottom: modalBottom
          });
          
          // Mostrar modal en móvil
          if (modalContent) {
            modalContent.innerHTML = html;
            console.log('HTML insertado en modal');
          } else {
            console.error('modalContent es null!');
          }
          
          if (modalOverlay) {
            modalOverlay.classList.add('active');
            console.log('Overlay activado');
          }
          
          if (modalBottom) {
            modalBottom.classList.add('active');
            console.log('Modal bottom activado');
          }
        } else {
          // Mostrar en el panel lateral en desktop
          const detail = document.getElementById('detalle');
          console.log('Mostrando en panel desktop');
          if (detail) {
            detail.innerHTML = html;
          }
        }
      }
      
      const first = document.querySelector('.postulante-item');
      if (first && window.innerWidth > 768) {
        console.log('Cargando primera postulación en desktop');
        mostrarDetalle(first);
      }
    </script>

<script>
  // Script para cerrar modal (disponible globalmente)
  function cerrarModalPostulacion() {
    const modalOverlay = document.getElementById('modalOverlay');
    const modalBottom = document.getElementById('modalBottom');
    if (modalOverlay && modalBottom) {
      modalOverlay.classList.remove('active');
      modalBottom.classList.remove('active');
    }
  }

  // Event listener para cerrar con overlay
  document.addEventListener('DOMContentLoaded', function() {
    const modalOverlay = document.getElementById('modalOverlay');
    if (modalOverlay) {
      modalOverlay.addEventListener('click', cerrarModalPostulacion);
    }
    
    const closeBtn = document.querySelector('.modal-close-custom');
    if (closeBtn) {
      closeBtn.addEventListener('click', cerrarModalPostulacion);
    }
  });
</script>
