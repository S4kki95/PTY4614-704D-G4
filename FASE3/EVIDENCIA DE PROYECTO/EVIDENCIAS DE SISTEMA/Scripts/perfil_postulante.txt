  <script>
    (function() {
      const btn = document.getElementById('copyCvUrlBtn');
      if (!btn) return;
      btn.addEventListener('click', async () => {
        const url = btn.getAttribute('data-url');
        try {
          await navigator.clipboard.writeText(url);
          btn.textContent = '¡Copiado!';
          setTimeout(() => btn.textContent = 'Copiar enlace', 1500);
        } catch {
          const input = document.createElement('textarea');
          input.value = url;
          document.body.appendChild(input);
          input.select();
          document.execCommand('copy');
          document.body.removeChild(input);
          btn.textContent = '¡Copiado!';
          setTimeout(() => btn.textContent = 'Copiar enlace', 1500);
        }
      });
    })();
  </script>

  <script>
    // UI: permite solo números y una K/k al final (máximo 9 caracteres); reemplaza DV si se escribe un 10º carácter
    (function(){
      const rutInput = document.querySelector('input[name="rut"]');
      if(!rutInput) return;
      rutInput.addEventListener('input', () => {
        let v = rutInput.value.toUpperCase().replace(/\s+/g, '');
        v = v.replace(/[\.-]/g, '');
        v = v.replace(/[^0-9K]/g, '');
        if (v.length > 0) {
          const kIndex = v.indexOf('K');
          if (kIndex !== -1 && kIndex !== v.length - 1) {
            v = v.replace(/K/g, '');
          }
        }
        if (v.length > 9) {
          const nuevoDV = v[v.length - 1];
          v = v.slice(0, 8) + nuevoDV;
        }
        rutInput.value = v;
      });
    })();
  </script>
