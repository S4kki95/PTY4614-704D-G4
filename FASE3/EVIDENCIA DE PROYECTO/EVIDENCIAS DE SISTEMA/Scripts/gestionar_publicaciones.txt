  <script>
  const IS_CAPACITADOR = (document.body.dataset.role === 'capacitador');
  
  // Delegación de eventos para cargar detalle
  document.addEventListener('click', (e) => {
    const item = e.target.closest('.anuncio-item');
    if (item && item.dataset.id) {
      const id = item.dataset.id;
      
      // Marcar item como activo
      document.querySelectorAll('.anuncio-item').forEach(n => n.classList.remove('active'));
      item.classList.add('active');
      
      mostrarDetalle(id);
    }
  });

  function mostrarDetalle(id) {
    const isMobile = window.innerWidth <= 768;
    
    fetch(`/portal/editar_publicacion/${id}/`, {
      headers: {"X-Requested-With": "XMLHttpRequest"}
    })
    .then(response => response.json())
    .then(data => {
      if (!data.success) {
        const htmlContent = data.html_form;
        
        if (isMobile) {
          // Mostrar en modal móvil
          const modalContent = document.getElementById('modalContentGestion');
          const modalOverlay = document.getElementById('modalOverlay');
          const modalBottom = document.getElementById('modalBottom');
          
          if (modalContent) {
            modalContent.innerHTML = htmlContent;
          }
          if (modalOverlay) {
            modalOverlay.classList.add('active');
          }
          if (modalBottom) {
            modalBottom.classList.add('active');
          }
          
          // Si es capacitador, deshabilitar inputs en el modal
          if (IS_CAPACITADOR) {
            modalContent.querySelectorAll('input, textarea, select, button[type="submit"]').forEach(el => {
              el.setAttribute('disabled', 'disabled');
            });
          }
          
          // Configurar formulario en modal
          setupForm(modalContent.querySelector('form'));
        } else {
          // Mostrar en panel desktop
          document.getElementById("detalle").innerHTML = htmlContent;
          
          // Si es capacitador, deshabilitar inputs
          if (IS_CAPACITADOR) {
            const container = document.getElementById("detalle");
            container.querySelectorAll('input, textarea, select, button[type="submit"]').forEach(el => {
              el.setAttribute('disabled', 'disabled');
            });
          }
          
          // Configurar formulario en desktop
          setupForm(document.querySelector("#detalle form"));
        }
      }
    });
  }

  function setupForm(form) {
    if (!form) return;
    
    form.onsubmit = function(e) {
      // Bloquear envío en capacitador (solo lectura)
      if (IS_CAPACITADOR) {
        e.preventDefault();
        return false;
      }
      
      e.preventDefault();
      fetch(form.action, {
        method: "POST",
        body: new FormData(form),
        headers: {"X-Requested-With": "XMLHttpRequest"}
      })
      .then(response => response.json())
      .then(data => {
        if (data.success) {
          window.location.reload();
        } else {
          const isMobile = window.innerWidth <= 768;
          if (isMobile) {
            document.getElementById('modalContentGestion').innerHTML = data.html_form;
            setupForm(document.querySelector('#modalContentGestion form'));
          } else {
            document.getElementById("detalle").innerHTML = data.html_form;
            setupForm(document.querySelector("#detalle form"));
          }
        }
      });
    };
  }

  // Auto-ajustar textareas
  document.addEventListener("input", function(e) {
    if (e.target.tagName.toLowerCase() === "textarea") {
      e.target.style.height = "auto";
      e.target.style.height = (e.target.scrollHeight) + "px";
    }
  });

  window.addEventListener("load", () => {
    document.querySelectorAll("textarea").forEach(t => {
      t.style.height = "auto";
      t.style.height = (t.scrollHeight) + "px";
    });
  });

  // Auto-ajustar textareas
  document.addEventListener("input", function(e) {
    if (e.target.tagName.toLowerCase() === "textarea") {
      e.target.style.height = "auto";
      e.target.style.height = (e.target.scrollHeight) + "px";
    }
  });

  window.addEventListener("load", () => {
    document.querySelectorAll("textarea").forEach(t => {
      t.style.height = "auto";
      t.style.height = (t.scrollHeight) + "px";
    });
  });

  // Cerrar modal
  function cerrarModalGestion() {
    const modalOverlay = document.getElementById('modalOverlay');
    const modalBottom = document.getElementById('modalBottom');
    if (modalOverlay && modalBottom) {
      modalOverlay.classList.remove('active');
      modalBottom.classList.remove('active');
    }
  }

  // Event listeners para cerrar modal
  document.addEventListener('DOMContentLoaded', function() {
    const modalOverlay = document.getElementById('modalOverlay');
    if (modalOverlay) {
      modalOverlay.addEventListener('click', cerrarModalGestion);
    }
    
    const closeBtn = document.querySelector('.modal-close-custom');
    if (closeBtn) {
      closeBtn.addEventListener('click', cerrarModalGestion);
    }
  });
  </script>
